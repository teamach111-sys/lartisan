%  COMMANDE \screenshot — Image encadrée avec hauteur uniforme
% ============================================================
\newcommand{\screenshot}[2]{%
  \begin{figure}[H]
    \centering
    \begin{tcolorbox}[
      enhanced,
      boxrule=0.4pt,
      colframe=grayrule,
      colback=white,
      arc=2pt,
      shadow={1.5mm}{-1.5mm}{0mm}{black!12},
      left=2pt, right=2pt, top=2pt, bottom=2pt,
      shrink tight,
      hbox % Fits box to image size
    ]
      \includegraphics[height=8cm, width=\textwidth, keepaspectratio]{#1}
    \end{tcolorbox}
    \caption{#2}
  \end{figure}%
}
